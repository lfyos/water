struct location_parameter
{
	component_number			:	i32,
	system_id_number			:	i32,
	component_workgroup_size	:	i32,
	system_id_workgroup_size	:	i32
};

@group(0) @binding(0)	var<storage,read_write> id_info			: array<id_information>;
@group(0) @binding(1)	var<storage,read_write> absolute_matrix	: array<mat4x4<f32>>;
@group(0) @binding(2)	var<storage,read> 		move_matrix		: array<mat4x4<f32>>;
@group(0) @binding(3)	var<storage,read> 		relative_matrix	: array<mat4x4<f32>>; 
@group(0) @binding(4)	var<storage,read> 		parent_id		: array<i32>;
@group(0) @binding(5)	var<uniform> 			location_par	: location_parameter;


@compute @workgroup_size(1)	fn compute_location_main(@builtin(global_invocation_id)global_id: vec3<u32>)
{	
	var component_id	:	i32;
	
	component_id=i32(global_id.z);
	component_id=i32(global_id.y)+location_par.component_workgroup_size*component_id;
	component_id=i32(global_id.x)+location_par.component_workgroup_size*component_id;
	
	if(component_id<location_par.component_number){
		var component_matrix=mat4x4<f32>(
				vec4<f32>(1.0,0.0,0.0,0.0),vec4<f32>(0.0,1.0,0.0,0.0),
				vec4<f32>(0.0,0.0,1.0,0.0),vec4<f32>(0.0,0.0,0.0,1.0));	
		for(var i=component_id;i>=0;i=parent_id[i])
			{component_matrix=relative_matrix[i]*move_matrix[i]*component_matrix;}
		absolute_matrix[component_id]=component_matrix;	
	}
}	

@compute @workgroup_size(1)	fn set_location_main(@builtin(global_invocation_id)global_id: vec3<u32>)
{
	var id_index_id	:	i32;
	
	id_index_id=i32(global_id.z);
	id_index_id=i32(global_id.y)+location_par.system_id_workgroup_size*id_index_id;
	id_index_id=i32(global_id.x)+location_par.system_id_workgroup_size*id_index_id;
	
	if(id_index_id<location_par.system_id_number)
		{id_info[id_index_id].matrix=absolute_matrix[id_info[id_index_id].component_id];}
}
