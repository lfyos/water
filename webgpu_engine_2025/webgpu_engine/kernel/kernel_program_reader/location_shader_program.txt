@group(0) @binding(0)	var<storage,read_write> id_info			:	array<id_information>;
@group(0) @binding(1)	var<storage,read_write> absolute_matrix	:	array<mat4x4<f32>>;
@group(0) @binding(2)	var<storage,read_write> camera_info		:	array<camera_information>;

@group(0) @binding(3)	var<storage,read> 		move_matrix		:	array<mat4x4<f32>>;
@group(0) @binding(4)	var<storage,read> 		relative_matrix	:	array<mat4x4<f32>>; 
@group(0) @binding(5)	var<storage,read> 		parent_id		:	array<i32>;
@group(0) @binding(6)	var<storage,read> 		location_flag	:	array<i32>;

override component_number			:	i32;
override system_id_number			:	i32;
override camera_number				:	i32;
override component_workgroup_size	:	i32;
override system_id_workgroup_size	:	i32;

@compute @workgroup_size(1)	fn compute_location_main(@builtin(global_invocation_id)global_id: vec3<u32>)
{	
	var component_id	:	i32;
	
	component_id=i32(global_id.z);
	component_id=i32(global_id.y)+component_workgroup_size*component_id;
	component_id=i32(global_id.x)+component_workgroup_size*component_id;
	
	if(component_id<component_number){
		if(location_flag[component_id]>0)
			{absolute_matrix[component_id]=move_matrix[component_id];}
		else{
			var component_matrix=relative_matrix[component_id]*move_matrix[component_id];	
			for(var i=parent_id[component_id];i>=0;i=parent_id[i]){
				component_matrix=move_matrix[i]*component_matrix;
				if(location_flag[i]>0)
					{break;}
				component_matrix=relative_matrix[i]*component_matrix;
			}
			absolute_matrix[component_id]=component_matrix;	
		}
	}
}
@compute @workgroup_size(1)	fn set_location_main(@builtin(global_invocation_id)global_id: vec3<u32>)
{
	var id_index_id		:	i32;
	
	id_index_id=i32(global_id.z);
	id_index_id=i32(global_id.y)+system_id_workgroup_size*id_index_id;
	id_index_id=i32(global_id.x)+system_id_workgroup_size*id_index_id;
	
	if(id_index_id<system_id_number){
		var my_component_id=id_info[id_index_id].component_id;
		id_info[id_index_id].matrix=absolute_matrix[my_component_id];
	}
	if(id_index_id<camera_number){
		var my_component_id	=camera_info[id_index_id].component_id;
		var my_matrix		=absolute_matrix[my_component_id];
		var my_eye_point	=my_matrix*vec4(0.0,0.0,camera_info[id_index_id].distance,1.0);

		camera_info[id_index_id].matrix		=my_matrix;
		camera_info[id_index_id].eye_point	=my_eye_point/my_eye_point.w;
		camera_info[id_index_id].near_value	=camera_info[id_index_id].near_value_ratio	*camera_info[id_index_id].distance;
		camera_info[id_index_id].far_value	=camera_info[id_index_id].far_value_ratio	*camera_info[id_index_id].distance;
	}
}
