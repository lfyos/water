struct vertex_to_fragment_struct
{
	@builtin(position)	vertex_position		:	vec4<f32>,
	@location(1)		vertex_id			:	vec4<f32>
};
struct id_target_struct
{
	@builtin(frag_depth)	depth	:	f32,
	@location(0) 			id_0	:	vec4<i32>,
	@location(1) 			id_1	:	vec4<i32>
};
struct color_target_struct
{
	@builtin(frag_depth)	depth	:	f32,
	@location(0) 			color	:	vec4<f32>
};

fn caculate_view_volume_distance()->f32
{
	var positive_matrix=target_info.projection_matrix;
		positive_matrix=target_info.negative_screen_move_matrix*positive_matrix;
		positive_matrix=target_info.screen_move_matrix_from*positive_matrix;
	
	var negative_matrix=target_info.negative_projection_matrix;
		negative_matrix=negative_matrix*target_info.screen_move_matrix;
		negative_matrix=negative_matrix*target_info.negative_screen_move_matrix_from;
	
	var center_point=vec4(0.0,0.0,0.0,1.0);
		center_point=target_info.camera_absolute_matrix*center_point;
		center_point=positive_matrix*center_point;
		center_point=center_point/center_point.w;
	
	var left_down_center_point=negative_matrix*vec4(-1.0,-1.0,center_point.z,1.0);
	var right_up_center_point =negative_matrix*vec4( 1.0, 1.0,center_point.z,1.0);
	
	left_down_center_point	=left_down_center_point/left_down_center_point.w;
	right_up_center_point	=right_up_center_point /right_up_center_point.w;

	return distance(right_up_center_point.xyz,left_down_center_point.xyz);
}

@vertex
fn vertex_main(
		@location(0) my_coord			:	vec4<f32>,
		@location(1) my_id				:	vec4<f32>
	)->vertex_to_fragment_struct
{
	var par_depth_start		=id_info.data[0].x;
	var par_depth_end		=id_info.data[0].y;
	var par_size			=id_info.data[0].z;
	var par_box_distance	=id_info.data[0].w;
	var position_par		=id_info.data[1];
	
	var world_coordinate_distance=caculate_view_volume_distance()*par_size;
	var model_scale_factor		 =world_coordinate_distance/par_box_distance;
	
	var p=target_info.camera_absolute_matrix*vec4(0.0,0.0,0.0,1.0);
		p=target_info.projection_matrix*vec4(p.xyz/p.w+my_coord.xyz*model_scale_factor,1.0);
		p=target_info.negative_screen_move_matrix*p;
		p=target_info.screen_move_matrix_from*p;
		p=p/p.w;
	var p0=target_info.camera_absolute_matrix*vec4(0.0,0.0, world_coordinate_distance/2.0,1.0);
		p0=target_info.projection_matrix*p0;
		p0=target_info.negative_screen_move_matrix*p0;
		p0=target_info.screen_move_matrix_from*p0;
		p0=p0/p0.w;
	var p1=target_info.camera_absolute_matrix*vec4(0.0,0.0,-world_coordinate_distance/2.0,1.0);
		p1=target_info.projection_matrix*p1;
		p1=target_info.negative_screen_move_matrix*p1;
		p1=target_info.screen_move_matrix_from*p1;
		p1=p1/p1.w;

	var depth=par_depth_start+(par_depth_end-par_depth_start)*((p.z-p0.z)/(p1.z-p0.z));

	p=target_info.negative_screen_move_matrix_from*vec4(p.xy+position_par.xy,depth,1.0);
	p=target_info.screen_move_matrix*p;

	var vf:vertex_to_fragment_struct;
	
	vf.vertex_id=my_id;
	vf.vertex_position=p;
	
	return vf;
};

@fragment
fn fragment_id_fun(vf:vertex_to_fragment_struct) -> id_target_struct
{
	var ret_val	:	id_target_struct;
	ret_val.id_0=vec4(id_info.component_system_id,
		i32(vf.vertex_id.x),i32(vf.vertex_id.y),(target_info.scene_id<<4)+0);
	ret_val.id_1=vec4(-1,-1,i32(vf.vertex_id.z),i32(vf.vertex_id.w));
	ret_val.depth=vf.vertex_position.z;
	return ret_val;
}

@fragment
fn fragment_face_fun(vf:vertex_to_fragment_struct) -> color_target_struct
{
	var ret_val:color_target_struct;
	
	ret_val.color=vec4<f32>(0.25,0.25,0.25,1.0);
	ret_val.depth=vf.vertex_position.z;
	
	return ret_val;
}
@fragment
fn fragment_edge_fun(vf:vertex_to_fragment_struct) -> color_target_struct
{
	var ret_val:color_target_struct;
	
	ret_val.color=vec4<f32>(0.0,1.0,0.0,1.0);
	ret_val.depth=vf.vertex_position.z-0.001;
	if(ret_val.depth<0.0)
		{ret_val.depth=0.0;}
	
	if(system_info.pickup_component_id==id_info.component_id){
		if(system_info.pickup_driver_id==id_info.driver_id){
			if(system_info.pickup_body_id==i32(vf.vertex_id.x)){
				if(system_info.pickup_face_id==i32(vf.vertex_id.y)){
					ret_val.color=vec4<f32>(1.0,0.0,0.0,1.0);
					ret_val.depth=0.0;
				}
			}
		}
	}
	return ret_val;
}
