struct vertex_to_fragment_struct
{
	@builtin(position)	vertex_position		:	vec4<f32>,	
	@location(1)		vertex_color_depth	:	vec4<f32>
};
struct color_target_struct
{
	@builtin(frag_depth)	depth	:	f32,
	@location(0) 			color	:	vec4<f32>
};
@vertex
fn vertex_main(
		@location(0) my_coord		:	vec4<f32>,
		@location(1) my_color_depth	:	vec4<f32>,
		@location(2) my_p0			:	vec4<f32>,
		@location(3) my_p1			:	vec4<f32>
	)->vertex_to_fragment_struct
{	
	var vf:vertex_to_fragment_struct;

	vf.vertex_position		=my_p0*(vec4(1.0,1.0,1.0,1.0)-my_coord)+my_p1*my_coord;
	vf.vertex_position		=target_info.projection_matrix*(id_info.matrix*vf.vertex_position);
	vf.vertex_color_depth	=my_color_depth;

	return vf;
}

@fragment
fn fragment_main(vf	:vertex_to_fragment_struct) -> color_target_struct
{
	var cts:color_target_struct;
	
	cts.color=vec4(vf.vertex_color_depth.rgb,1.0);
	cts.depth=vf.vertex_color_depth.w;
	
	if(system_info.millisecond>500)
		{discard;}
		
	return cts;
}
