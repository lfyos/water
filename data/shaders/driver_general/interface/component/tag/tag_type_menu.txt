function(component_name,component_id,scene)
{
	scene.component_event_processor[component_id]={
		type_base_id		:	0,
		pickupmouseselect	:	function(event,component_id,scene)
		{
			var ep=scene.component_event_processor[component_id];

			var type_id=-1,function_id=Math.floor(ep.y*12.0);
			switch(function_id){
			case 0:	
				return;
			case 1:
				type_id=0;break;
			case 2:
				type_id=1;break;
			case 3:
				type_id=2;break;
			case 4:	
				return;
			case 5:
				type_id=3;break;
			case 6:
				type_id=4;break;
			case 7:
				type_id=5;break;
			case 8:	
				return;
			case 9:
				type_id=6;break;
			case 10:
				type_id=7;break;
			case 11:
				type_id=8;break;
			}
			type_id+=ep.type_base_id
			
			var type_name=[
			/*	00-02	*/	"absolute_distance",	"projection_distance",	"projection_angle",
			
			/*	03-11	*/	"global_direction_x",	"global_direction_y",	"global_direction_z",	
							"local_direction_x",	"local_direction_y",	"local_direction_z",
							"view_direction_x",		"view_direction_y",		"view_direction_z",	
			/*	12-20	*/	"global_plane_yz",		"global_plane_zx",		"global_plane_xy",
							"local_plane_yz",		"local_plane_zx",		"local_plane_xy",
							"view_plane_yz",		"view_plane_zx",		"view_plane_xy",
			/*	21-29	*/	"global_angle_x",		"global_angle_y",		"global_angle_z",
							"local_angle_x",		"local_angle_y",		"local_angle_z",
							"view_angle_x",			"view_angle_y",			"view_angle_z",
			/*	30-38	*/	"global_angle_yz",		"global_angle_zx",		"global_angle_xy",
							"local_angle_yz",		"local_angle_zx",		"local_angle_xy",
							"view_angle_yz",		"view_angle_zx",		"view_angle_xy",
			];
			if((type_id<0)||(type_id>=type_name.length))
				type_id=0;
			
			var tag_ep=scene.operate_component.get_component_event_processor("distance_tag_component");
			if(tag_ep.pickup_tag_id>=0)
				scene.caller.call_server_component(tag_ep.component_id,"all",
					[["operation","type"],["id",tag_ep.pickup_tag_id],["type",type_name[type_id]]]);
		}
	}
}
